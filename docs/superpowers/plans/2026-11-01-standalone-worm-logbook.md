```js
// ISO timestamp for November <d>, 2026 at <h>:00 local time.
export const T = (d, h = 9) => new Date(2026, 10, d, h).toISOString();

export function state({ batches = [], events = [], settings = {} } = {}) {
  const S = { batches: {}, events: {}, settings, bin: { name: "Worm bin" } };

  events.forEach((e, i) => { const id = e.id || "e" + i; S.events[id] = { ...e, id, day: e.day || dayKey(e.ts) } });

}

export const check = (d, binTemp, rows = [], extra = {}) => ({ type: "check", ts: T(d, extra.h || 9), binTemp, batches: rows, ...extra });
```

```js
// ---------- state helpers ----------
export const settingsOf = S => ({ ...DEF, ...(S.settings || {}) });
export const batches = S => Object.values(S.batches).sort((a, b) => (b.num || 0) - (a.num || 0));
export const events = S => Object.values(S.events).sort((a, b) => b.ts < a.ts ? -1 : 1);
export const active = S => batches(S).filter(b => b.status !== "used");
export const bName = b => b ? (b.nickname ? b.nickname + " (#" + b.num + ")" : "Batch #" + b.num) : "Unknown batch";
export const bShort = b => b ? (b.nickname || "Batch #" + b.num) : "?";

// readings for a batch: from check events, newest first
export function batchReadings(S, bid) {
  const out = [];
  for (const e of events(S)) {
    if (e.type !== "check" || !Array.isArray(e.batches)) continue;
    const r = e.batches.find(x => x.id === bid);
    if (r) out.push({ ts: e.ts, day: e.day || dayKey(e.ts), temp: r.temp, smell: r.smell, stirred: r.stirred, bin: e.binTemp });
  }
  return out;
}
export function lastStir(S, b) {
  let best = null;
  for (const e of events(S)) {
    if (e.type === "check" && Array.isArray(e.batches)) { const r = e.batches.find(x => x.id === b.id); if (r && r.stirred && (!best || e.ts > best)) best = e.ts }
    if (e.type === "scraps" && e.batchId === b.id && e.mixed && (!best || e.ts > best)) best = e.ts;
  }
  return best;
}
export function lastSmell(S, b) { const r = batchReadings(S, b.id).find(x => x.smell); return r ? r.smell : null }
export function binTempOnDay(S, day) {
  for (const e of events(S)) if (e.type === "check" && (e.day || dayKey(e.ts)) === day && e.binTemp != null) return e.binTemp;
  return null;
}

// Readiness: latest reading per day since resting started; count consecutive qualifying days from newest.
export function readiness(S, b) {
  const st = settingsOf(S);
  const since = b.restStartedAt ? dayKey(b.restStartedAt) : null;
  const perDay = new Map();
  for (const r of batchReadings(S, b.id)) { // newest first, so first seen per day = latest
    if (r.temp == null) continue;
    if (since && r.day < since) continue;
    if (!perDay.has(r.day)) perDay.set(r.day, { ...r, bin: r.bin != null ? r.bin : binTempOnDay(S, r.day) });
  }
  const days = [...perDay.values()].sort((a, b) => a.day < b.day ? 1 : -1);
  let count = 0, lo = Infinity, hi = -Infinity, reason = "";
  for (const r of days) {
    if (r.bin == null) { reason = reason || "Log the worm bin temp with each batch reading."; break }
    if (r.temp > st.maxTemp) { reason = reason || `Still warm: ${r.temp}°C (limit ${st.maxTemp}°C).`; break }
    if (Math.abs(r.temp - r.bin) > st.spread) { reason = reason || `${r.temp}°C vs bin ${r.bin}°C — not cooled to bin temp yet.`; break }
    if (r.smell === "sour" || r.smell === "ammonia") { reason = reason || "Smelled " + r.smell + " — add browns and stir."; break }
    const nlo = Math.min(lo, r.temp), nhi = Math.max(hi, r.temp);
    if (nhi - nlo > st.spread) { reason = reason || "Temps not steady yet."; break }
    lo = nlo; hi = nhi; count++;
    if (count >= st.readings) break;
  }
  const latest = days[0] || null;
  if (!days.length) reason = "No temp readings since it started resting.";
  return { count: Math.min(count, st.readings), need: st.readings, ready: count >= st.readings, latest, reason: count >= st.readings ? "" : reason };
}
```

```js
export function alerts(S, now = new Date()) {
  const st = settingsOf(S), out = [];
  const checks = events(S).filter(e => e.type === "check");
  const lastCheck = checks[0];
  const lastBin = checks.find(e => e.binTemp != null);
  if (lastBin && lastBin.binTemp > st.maxTemp) out.push({ k: "bad", ic: "🌡️", t: `Worm bin is hot: ${lastBin.binTemp}°C`, d: "ANC get stressed above ~30°C. Move to shade, mist lightly, add fresh bedding." });
  if (lastCheck && lastCheck.activity === "escaping") out.push({ k: "bad", ic: "🚨", t: "Worms trying to escape", d: "Usually too wet, too hot, or unsafe food. Check moisture and smell; remove fresh food." });
  if (lastCheck && lastCheck.moisture === "wet") out.push({ k: "warn", ic: "💧", t: "Bin is wet", d: "Add dry browns (shredded cardboard/paper) and hold off misting." });
  if (lastCheck && lastCheck.moisture === "dry") out.push({ k: "warn", ic: "🏜️", t: "Bin is dry", d: "Mist until it feels like a wrung-out sponge." });
  if (lastCheck && lastCheck.activity === "sluggish") out.push({ k: "warn", ic: "🐌", t: "Worms are sluggish", d: "Check temp, moisture and smell on the next check." });
  if (!lastCheck) out.push({ k: "info", ic: "📝", t: "No bin check yet", d: "Log your first daily check to start tracking.", act: "check" });
  else if (daysBetween(lastCheck.ts, now) >= st.checkinDays) out.push({ k: "warn", ic: "⏰", t: `No check-in for ${daysBetween(lastCheck.ts, now)} days`, d: "Do a quick bin check.", act: "check" });
  for (const b of active(S)) {
    if (b.status === "resting") {
      const r = readiness(S, b);
      if (r.ready) out.push({ k: "good", ic: "✅", t: `${bName(b)} looks ready`, d: `${r.need} steady, safe readings in a row. Confirm when it smells earthy.`, act: "ready:" + b.id, av: b.avatar });
      else {
        const has = batchReadings(S, b.id).some(x => x.day === today(now) && x.temp != null);
        if (!has) out.push({ k: "info", ic: "🌡️", t: `Log today's temp for ${bName(b)}`, d: `Readiness ${r.count}/${r.need}.`, act: "check", av: b.avatar });
      }
    }
    if (b.status === "filling" || b.status === "resting") {
      const ls = lastStir(S, b) || b.createdAt;
      const d = daysBetween(ls, now);
      if (d >= st.stirDays) out.push({ k: "warn", ic: "🥄", t: `Stir ${bName(b)}`, d: `Last stirred ${ago(lastStir(S, b), now)}. Stirring keeps it aerobic.`, act: "check", av: b.avatar });

      if (sm === "sour" || sm === "ammonia") out.push({ k: "warn", ic: "👃", t: `${bName(b)} smelled ${sm}`, d: "Add browns (cardboard, paper, dry leaves) and stir.", av: b.avatar });
    }
  }
  const all = active(S);
  const feeding = all.filter(b => b.status === "feeding"), filling = all.filter(b => b.status === "filling"), pipeline = all.filter(b => b.status === "resting" || b.status === "ready");
  if (!filling.length && batches(S).length) out.push({ k: "info", ic: "🪣", t: "No batch is filling", d: "Start a new batch for your scraps.", act: "newbatch" });
  if (feeding.length && !pipeline.length && filling.length) out.push({ k: "info", ic: "📦", t: "Nothing resting for next feeding", d: "Consider marking your filling batch full so it has time to cool before the worms run out." });

}
```

```js
// The only Firestore data access. Everything lives under users/{uid}/.
// put/del return Firestore's promise, which only settles once the server has the write;
// callers must not await it (offline it stays pending until the phone reconnects).
import { collection, doc, setDoc, deleteDoc, onSnapshot } from "firebase/firestore";
import { db } from "./firebase.js";

let uid = null;
const ref = (coll, id) => doc(db, "users", uid, coll, id);

export function put(coll, id, obj) {
  if (!uid) return Promise.reject(new Error("Not signed in"));
  return setDoc(ref(coll, id), obj);
}

export function del(coll, id) {
  if (!uid) return Promise.reject(new Error("Not signed in"));
  return deleteDoc(ref(coll, id));
}

// onData(kind, data): kind is "batches" | "events" (map of id → doc with id) or "settings" | "bin" (doc).
// onLoaded(): both collections have arrived from the server. onSlow(): not loaded after 10 s.
export function subscribe(userId, { onData, onLoaded, onSlow, onError }) {
  uid = userId;
  const loaded = { batches: false, events: false };
  const timer = setTimeout(() => { if (!(loaded.batches && loaded.events)) onSlow() }, 10000);
  const err = e => { clearTimeout(timer); onError(e) };
  const toMap = s => { const m = {}; s.docs.forEach(x => { m[x.id] = { ...x.data(), id: x.id } }); return m };
  const coll = kind => onSnapshot(collection(db, "users", userId, kind), { includeMetadataChanges: true }, s => {
    onData(kind, toMap(s));
    if (!s.metadata.fromCache) { loaded[kind] = true; if (loaded.batches && loaded.events) { clearTimeout(timer); onLoaded() } }
  }, err);
  const single = kind => onSnapshot(ref("config", kind), s => { if (s.exists()) onData(kind, s.data()) }, err);
  const unsubs = [coll("batches"), coll("events"), single("settings"), single("bin")];
  return () => { clearTimeout(timer); unsubs.forEach(u => u()); uid = null };
}
```
